CREATE TABLE plants (
    plant_id INT UNSIGNED NOT NULL AUTO_INCREMENT,
    wand_id  VARCHAR(20) NOT NULL,
    genus    VARCHAR(255),
    species  VARCHAR(255),
    gbif_key VARCHAR(15),
    family_id INT UNSIGNED NOT NULL,
    PRIMARY KEY (plant_id),
    UNIQUE (wand_id),
    CONSTRAINT fk_plant_family
        FOREIGN KEY (family_id)
        REFERENCES families(family_id)
        ON UPDATE CASCADE
        ON DELETE RESTRICT
);
